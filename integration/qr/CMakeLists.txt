cmake_minimum_required(VERSION 3.20)
project(draftor_qr LANGUAGES C CXX)
option(QR_SANITIZE "Standalone QR ASan/UBSan" OFF)
if(QR_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(draftor_qr STATIC src/qr.cpp vendor/nayuki/qrcodegen.c)
target_include_directories(draftor_qr PUBLIC include PRIVATE vendor/nayuki)
target_compile_features(draftor_qr PUBLIC cxx_std_17)
target_compile_options(draftor_qr PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-usage)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 enable_testing()
 foreach(name qr_contract qr_fixture)
  add_executable(${name} tests/${name}.cpp)
  target_link_libraries(${name} PRIVATE draftor_qr)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endforeach()
 add_test(NAME qr_contract COMMAND qr_contract)
endif()
