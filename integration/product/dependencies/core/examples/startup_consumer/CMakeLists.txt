cmake_minimum_required(VERSION 3.20)
project(HanIMEStartupConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(HANIME_CORE_PATH "" CACHE PATH "Source tree of the new authoritative core")
set(BASIC_HOST_PATH "" CACHE PATH "Unchanged Basic Host v0.5 source tree")
set(HANIME_RELEASE_PIN_DIR "" CACHE PATH "Host-published, firmware-accepted release_pin.hpp")
option(STARTUP_SANITIZERS "ASan and UBSan, with strict warnings retained" OFF)
foreach(required HANIME_CORE_PATH BASIC_HOST_PATH HANIME_RELEASE_PIN_DIR)
  if(NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "Set ${required} to its explicit existing directory")
  endif()
endforeach()
function(require_basic_file relative expected)
  file(SHA256 "${BASIC_HOST_PATH}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Basic Host v0.5 frozen source mismatch: ${relative}")
  endif()
endfunction()
require_basic_file(src/composition.cpp be18d3c8b3a920470d2ab1c0c12c43fea7aa4be74cdab014b6f0501c4bfa9962)
require_basic_file(src/submission.cpp ff11354c2c5f068bf9c816fe5d30bc4131d6cd696b4c70913360bef70c7ac939)
require_basic_file(src/navigation.cpp 3b95305185a1dcd2f75a0f2a18ff35097b37332fe9450f22c249d87dbae0ef7e)
require_basic_file(src/navigation_syllables.inc 6aeeffc0040ba95233b6b3502ac06a7ade1a601da87f6e674fa5c43b6645ad6b)
require_basic_file(include/hanime_extension/extension.hpp 8350fcca5674e745b792ca88d11f0929b3e3ee46df92845c6012249d7c4eabc9)
require_basic_file(include/hanime_extension/submission.hpp a969e174e7e34a7bc32840401ac642e79cbfedf50dd0578c48ed71cf86dba855)
require_basic_file(include/hanime_extension/navigation.hpp d30492714408494bee2be11c82ba3bd2b6f704f9b1e8c40064369834a155a38a)
set(HANIME_BUILD_HOST_ADAPTERS OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_FIXTURES OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(HANIME_ENABLE_SANITIZERS ${STARTUP_SANITIZERS} CACHE BOOL "" FORCE)
add_subdirectory("${HANIME_CORE_PATH}" core)
# Deliberately do not add Basic Host's top-level CMake, which pins core 0.8.0.
# Compile its three exact, unchanged consumer units against this one core.
add_library(startup_basic_host STATIC
  "${BASIC_HOST_PATH}/src/composition.cpp"
  "${BASIC_HOST_PATH}/src/submission.cpp"
  "${BASIC_HOST_PATH}/src/navigation.cpp")
target_include_directories(startup_basic_host PUBLIC "${BASIC_HOST_PATH}/include")
target_link_libraries(startup_basic_host PUBLIC HanIME::Core)
target_compile_options(startup_basic_host PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-usage)
add_executable(startup_consumer main.cpp)
target_include_directories(startup_consumer PRIVATE "${HANIME_RELEASE_PIN_DIR}")
target_link_libraries(startup_consumer PRIVATE startup_basic_host)
target_compile_options(startup_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-usage)
