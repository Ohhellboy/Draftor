cmake_minimum_required(VERSION 3.20)
project(HanIMEQueryConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(HANIME_CORE_PATH "" CACHE PATH "Authoritative core source")
set(HANIME_RELEASE_PIN_DIR "" CACHE PATH "Trusted host-generated pin")
option(QUERY_SANITIZERS "Address and undefined behavior checks" OFF)
if(NOT IS_DIRECTORY "${HANIME_CORE_PATH}" OR NOT EXISTS "${HANIME_RELEASE_PIN_DIR}/release_pin.hpp")
  message(FATAL_ERROR "Explicit core and trusted pin directories required")
endif()
set(HANIME_BUILD_HOST_ADAPTERS OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_FIXTURES OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(HANIME_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(HANIME_ENABLE_SANITIZERS ${QUERY_SANITIZERS} CACHE BOOL "" FORCE)
add_subdirectory("${HANIME_CORE_PATH}" core)
add_executable(query_consumer main.cpp allocation_probe.cpp)
if(QUERY_SANITIZERS)
  target_compile_definitions(query_consumer PRIVATE QUERY_PROBE_DISABLED=1)
endif()
target_include_directories(query_consumer PRIVATE "${HANIME_RELEASE_PIN_DIR}")
target_link_libraries(query_consumer PRIVATE HanIME::Core)
target_compile_options(query_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-usage)
