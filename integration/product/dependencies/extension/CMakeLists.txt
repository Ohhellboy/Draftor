cmake_minimum_required(VERSION 3.20)
project(HanIMEInputCandidate VERSION 0.8.4 LANGUAGES CXX)
set(HANIME_HOST_CANDIDATE_VERSION "0.8.4-rc.1")
if(NOT TARGET HanIME::Core)
  set(EXTENSION_CORE_DIRECTORY "" CACHE PATH "Unchanged Core 0.13.1-rc.1 source")
  if(NOT EXISTS "${EXTENSION_CORE_DIRECTORY}/CMakeLists.txt")
    message(FATAL_ERROR "Provide the fixed RC1 source in EXTENSION_CORE_DIRECTORY")
  endif()
  set(HANIME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(HANIME_BUILD_FIXTURES OFF CACHE BOOL "" FORCE)
  set(HANIME_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(HANIME_BUILD_HOST_ADAPTERS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${EXTENSION_CORE_DIRECTORY}" core)
endif()
add_library(hanime_extension STATIC src/composition.cpp src/submission.cpp src/segment_planner.cpp)
target_include_directories(hanime_extension PUBLIC include)
target_compile_features(hanime_extension PUBLIC cxx_std_17)
target_compile_options(hanime_extension PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(hanime_extension PUBLIC HanIME::Core)
add_library(HanIME::Extension ALIAS hanime_extension)
add_library(hanime_navigation STATIC src/navigation.cpp)
target_compile_options(hanime_navigation PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(hanime_navigation PUBLIC HanIME::Extension)
add_library(HanIME::Navigation ALIAS hanime_navigation)
