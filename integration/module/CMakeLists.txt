cmake_minimum_required(VERSION 3.20)
project(draftor_usb_transfer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(USB_SANITIZE "ASan + UBSan" OFF)
set(EXCHANGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../shared" CACHE PATH "Single maintained TXT/session source")
if(USB_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
add_subdirectory("${EXCHANGE_ROOT}" exchange)
option(USB_ENABLED "Build USB controllers/platform tests (neutral exchange remains available)" ON)
if(NOT USB_ENABLED)
  return()
endif()
add_library(draftor_usb STATIC src/common/export.cpp src/common/inbox.cpp platform/shared/media_bridge.cpp)
target_include_directories(draftor_usb PUBLIC include platform/shared)
target_link_libraries(draftor_usb PUBLIC draftor_exchange)
target_compile_options(draftor_usb PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Threads REQUIRED)
add_library(usb_host_reference STATIC host/persistent_store.cpp)
target_include_directories(usb_host_reference PUBLIC host)
target_link_libraries(usb_host_reference PUBLIC draftor_usb)
target_compile_options(usb_host_reference PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(usb_tests tests/usb_tests.cpp)
target_include_directories(usb_tests PRIVATE "${EXCHANGE_ROOT}/tests")
target_link_libraries(usb_tests PRIVATE usb_host_reference Threads::Threads)
target_compile_options(usb_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(case export_volume request_lifetime inbox_ownership import_text transactions budget receipt_confirmation)
  add_test(NAME USB_${case} COMMAND usb_tests ${case} "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${case}")
  set_tests_properties(USB_${case} PROPERTIES TIMEOUT 60)
endforeach()
add_executable(usb_msc_callback_tests tests/msc_callbacks.cpp platform/esp32s3/tinyusb_msc.cpp)
target_include_directories(usb_msc_callback_tests PRIVATE tests/sdk_shim platform/esp32s3 "${EXCHANGE_ROOT}/tests")
target_link_libraries(usb_msc_callback_tests PRIVATE draftor_usb)
target_compile_options(usb_msc_callback_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME USB_msc_callbacks COMMAND usb_msc_callback_tests)
set_tests_properties(USB_msc_callbacks PROPERTIES TIMEOUT 30)

add_subdirectory(tests/s3_fat)
