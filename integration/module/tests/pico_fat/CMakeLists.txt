cmake_minimum_required(VERSION 3.20)
project(pico_real_fat_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MODULE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(CANDIDATE_ROOT "${MODULE_ROOT}/.." ABSOLUTE)
set(SDFAT_ROOT "" CACHE PATH "Pinned Arduino-Pico 6.0.0 SdFat src directory")
if(NOT EXISTS "${SDFAT_ROOT}/FatLib/FatFile.cpp")
 message(FATAL_ERROR "Set SDFAT_ROOT to the pinned real SdFat src directory")
endif()
option(PICO_FAT_SANITIZE "ASan + UBSan" OFF)
if(PICO_FAT_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
file(GLOB FAT_SOURCE CONFIGURE_DEPENDS "${SDFAT_ROOT}/FatLib/*.cpp" "${SDFAT_ROOT}/common/*.cpp")
add_library(real_sdfat STATIC ${FAT_SOURCE})
target_include_directories(real_sdfat PUBLIC "${SDFAT_ROOT}")
target_compile_definitions(real_sdfat PUBLIC ENABLE_ARDUINO_FEATURES=0 ENABLE_ARDUINO_SERIAL=0 ENABLE_ARDUINO_STRING=0 USE_UTF8_LONG_NAMES=1 USE_BLOCK_DEVICE_INTERFACE=1 SPI_DRIVER_SELECT=3)
target_compile_options(real_sdfat PUBLIC -include "${CMAKE_CURRENT_SOURCE_DIR}/host_compat.hpp")
add_executable(pico_fat_tests fat_tests.cpp "${MODULE_ROOT}/platform/pico/sdfat_inbox.cpp" "${MODULE_ROOT}/src/common/inbox.cpp" "${CANDIDATE_ROOT}/shared/src/txt.cpp")
target_include_directories(pico_fat_tests PRIVATE "${MODULE_ROOT}/include" "${MODULE_ROOT}/platform/pico" "${CANDIDATE_ROOT}/shared/include")
target_link_libraries(pico_fat_tests PRIVATE real_sdfat)
target_compile_options(pico_fat_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
foreach(case roundtrip full corrupt lifecycle)
 add_test(NAME PICO_REAL_FAT_${case} COMMAND pico_fat_tests ${case})
 set_tests_properties(PICO_REAL_FAT_${case} PROPERTIES TIMEOUT 30)
endforeach()

add_executable(pico_msc_tests ../pico_msc/callback_tests.cpp "${MODULE_ROOT}/platform/pico/tinyusb_msc.cpp" "${MODULE_ROOT}/platform/shared/media_bridge.cpp")
target_include_directories(pico_msc_tests PRIVATE "${MODULE_ROOT}/tests/pico_msc/shim" "${MODULE_ROOT}/include" "${MODULE_ROOT}/platform/pico" "${MODULE_ROOT}/platform/shared" "${CANDIDATE_ROOT}/shared/include")
target_compile_options(pico_msc_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME PICO_MSC_ADAPTER_callbacks COMMAND pico_msc_tests)
set_tests_properties(PICO_MSC_ADAPTER_callbacks PROPERTIES TIMEOUT 30)
