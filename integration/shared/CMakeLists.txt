cmake_minimum_required(VERSION 3.20)
project(draftor_exchange_step2a LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EXCHANGE_SANITIZE "Run focused Host tests with ASan and UBSan" OFF)
set(DRAFTOR_R2_SOURCE "" CACHE PATH "Read-only Phase2C_R2/source; omit for standalone H tests")
if(EXCHANGE_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "This sanitizer configuration requires Clang or GCC")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_library(draftor_exchange STATIC src/txt.cpp src/export_session.cpp src/import_job.cpp)
target_include_directories(draftor_exchange PUBLIC include)
target_compile_options(draftor_exchange PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(h_tests tests/h_tests.cpp)
target_link_libraries(h_tests PRIVATE draftor_exchange)
target_compile_options(h_tests PRIVATE -Wall -Wextra -Wpedantic)
foreach(case txt_boundaries txt_splits txt_invalid txt_backpressure txt_semantics
    session_preparing snapshot_ownership session_identity input_model lifecycle_100 module_budget)
  add_test(NAME H_${case} COMMAND h_tests ${case})
  set_tests_properties(H_${case} PROPERTIES LABELS H TIMEOUT 30)
endforeach()
# The two independent review programs are preserved byte-for-byte. One target
# per supplied program, not one test per assertion/sequence.
foreach(case lease_retry input_modifier)
  add_executable(review_${case} tests/review_reproducers/${case}_repro.cpp)
  target_include_directories(review_${case} PRIVATE tests)
  target_link_libraries(review_${case} PRIVATE draftor_exchange)
  target_compile_options(review_${case} PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME REVIEW_${case} COMMAND review_${case})
  set_tests_properties(REVIEW_${case} PROPERTIES LABELS "review;H" TIMEOUT 30)
endforeach()
if(DRAFTOR_R2_SOURCE)
  include(cmake/r2_host.cmake)
endif()

add_executable(transfer_contract tests/transfer_contract.cpp)
target_link_libraries(transfer_contract PRIVATE draftor_exchange)
target_compile_options(transfer_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME H_transfer_contract COMMAND transfer_contract)
set_tests_properties(H_transfer_contract PROPERTIES TIMEOUT 30)
