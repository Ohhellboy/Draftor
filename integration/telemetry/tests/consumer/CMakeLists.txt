cmake_minimum_required(VERSION 3.16)
project(ExternalTelemetryConsumer LANGUAGES CXX)
if(NOT TELEMETRY_SOURCE)
  message(FATAL_ERROR "Set TELEMETRY_SOURCE to the review package root")
endif()
# Deliberately request instrumented module tests too: public target must remain clean.
set(TELEMETRY_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${TELEMETRY_SOURCE}" telemetry)
get_target_property(policy draftor_telemetry INTERFACE_COMPILE_OPTIONS)
get_target_property(link_policy draftor_telemetry INTERFACE_LINK_OPTIONS)
if(policy OR link_policy)
  message(FATAL_ERROR "Consumer inherited private telemetry policy: ${policy}; ${link_policy}")
endif()
add_executable(external_consumer main.cpp)
target_link_libraries(external_consumer PRIVATE draftor_telemetry)
enable_testing()
add_test(NAME consumer_semantics COMMAND external_consumer)
